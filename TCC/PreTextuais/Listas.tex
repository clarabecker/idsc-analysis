\begin{simbolos}
	\item[$X$] Matriz ou vetor de características de entrada (atributos preditores do IDSC)
	\item[$Y$] Variável alvo (faixas de desempenho do IDEB)
	\item[$\hat{f}(X)$] Função estimada pelo modelo de aprendizado de máquina
	\item[$\hat{y}$] Valor predito pelo modelo
	\item[$TP$] Verdadeiros Positivos (\textit{True Positives})
	\item[$TN$] Verdadeiros Negativos (\textit{True Negatives})
	\item[$FP$] Falsos Positivos (\textit{False Positives})
	\item[$FN$] Falsos Negativos (\textit{False Negatives})
	\item[$F1$] Métrica $F_1$-\textit{score} (média harmônica entre precisão e \textit{recall})
	\item[$K$] Número total de classes no problema multiclasse
	\item[$c$] Classe específica avaliada no cenário multiclasse
	\item[$r$] Coeficiente de correlação linear de Pearson
	\item[$L_1, L_2$] Termos de regularização de normas L1 e L2
	\item[$\sum$] Símbolo de somatório utilizado nas equações de média e métricas
\end{simbolos}
